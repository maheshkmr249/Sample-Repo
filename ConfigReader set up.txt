public class configReader {
private static Properties prop;
public static void loadConfig(){
String env = System.getProperty("env", "qa");
prop = new Properties();
String filePath = "src/test/resourses/config/"+env+".Properties";
FileInputStream fis = new FileInputStream(filePath);
prop.load(fis);
sopln("loaded environment is: "+env);
}
public static String getProperty(String key){
return prop.getProperty(key);
}
}

public class BaseClass{
WebDriver driver;
@BeforeMethod
public void  setup()
{
configReader.loadConfig();
String browser = configReader.getProperty("browser");
if(browser.equalIgnoreCase("Chrome"))
{
driver = new ChromeDriver();
}
driver.manage().window().maximize();
driver.get(configReader.getProperty("baseUrl"));
}
@AfterMethod
public void tearDown(){
if(driver!=null)
{
driver.quit();
}
}
}

package pages;

public class LoginPage {

WebDriver driver;
// Locators
By username = By.id("user-name");
By password = By.id("password");
By loginButton = By.id("login-button");
By homePageTitle = By.className("title");

// Constructor
public LoginPage(WebDriver driver) {
this.driver = driver;
}
// Enter username
public void enterUsername(String user) {
driver.findElement(username).sendKeys(user);
}
// Enter password
public void enterPassword(String pass) {
driver.findElement(password).sendKeys(pass);
}
// Click login button
public void clickLogin() {
driver.findElement(loginButton).click();
}
// Complete login action
public void login(String user, String pass) {
enterUsername(user);
enterPassword(pass);
clickLogin();
}
// Get home page title
public String getHomePageTitle() {
return driver.findElement(homePageTitle).getText();
}
}
